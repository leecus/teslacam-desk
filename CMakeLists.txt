cmake_minimum_required(VERSION 3.24)
project(TeslaCamManager VERSION 0.2.1 LANGUAGES CXX)
if(MSVC AND DEFINED TESLACAM_MSVC_DEPS_PREFIX)
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "${TESLACAM_MSVC_DEPS_PREFIX}")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Sql Multimedia MultimediaWidgets Test LinguistTools)
add_library(camcore src/library.cpp src/library.h src/media.cpp src/media.h)
target_include_directories(camcore PUBLIC "${CMAKE_SOURCE_DIR}/src")
target_link_libraries(camcore PUBLIC Qt6::Core Qt6::Sql)
target_compile_definitions(camcore PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  add_compile_options(/utf-8 /W4)
  target_compile_options(camcore PRIVATE /utf-8 /W4)
endif()
qt_add_executable(TeslaCamManager WIN32 src/main.cpp src/window.cpp src/window.h src/player.cpp src/player.h src/theme.cpp src/theme.h src/i18n.cpp src/i18n.h)
target_link_libraries(TeslaCamManager PRIVATE camcore Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets)
target_compile_definitions(TeslaCamManager PRIVATE TESLACAM_VERSION="${PROJECT_VERSION}")
if(WIN32)
  target_sources(TeslaCamManager PRIVATE src/app.rc)
  target_include_directories(TeslaCamManager PRIVATE "${CMAKE_SOURCE_DIR}")
endif()
qt_add_resources(TeslaCamManager "app_icons" PREFIX "/icons" BASE "${CMAKE_SOURCE_DIR}/assets"
  FILES assets/teslacam.svg)
set_target_properties(TeslaCamManager PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")
# Chinese is the source language; English lives in the compiled translation, embedded so the
# executable stays self-contained.
qt_add_lrelease(TeslaCamManager TS_FILES "${CMAKE_SOURCE_DIR}/i18n/teslacam_en.ts" QM_FILES_OUTPUT_VARIABLE teslacam_translations)
qt_add_resources(TeslaCamManager "translations" PREFIX "/i18n" BASE "${CMAKE_CURRENT_BINARY_DIR}" FILES ${teslacam_translations})
enable_testing()
qt_add_executable(camcore_tests tests/core_tests.cpp)
set_target_properties(camcore_tests PROPERTIES WIN32_EXECUTABLE FALSE)
target_link_libraries(camcore_tests PRIVATE camcore Qt6::Test)
add_test(NAME library_safety COMMAND camcore_tests -o "${CMAKE_BINARY_DIR}/core-test-results.txt,txt")
qt_add_executable(camui_tests tests/ui_tests.cpp src/window.cpp src/window.h src/player.cpp src/player.h src/theme.cpp src/theme.h src/i18n.cpp src/i18n.h)
set_target_properties(camui_tests PROPERTIES WIN32_EXECUTABLE FALSE)
target_link_libraries(camui_tests PRIVATE camcore Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets Qt6::Test)
add_test(NAME ui_lifecycle COMMAND camui_tests -o "${CMAKE_BINARY_DIR}/ui-test-results.txt,txt")
set_tests_properties(ui_lifecycle PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
